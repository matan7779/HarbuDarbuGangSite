async function sha256(text){
  const bytes=new TextEncoder().encode(text);
  const digest=await crypto.subtle.digest("SHA-256",bytes);
  return Array.from(new Uint8Array(digest)).map(b=>b.toString(16).padStart(2,"0")).join("");
}

async function rpc(name,args={}){
  if(!db) throw new Error("Supabase is not configured.");
  const {data,error}=await db.rpc(name,args);
  if(error) throw error;
  return data;
}

async function verifyAccess(code){
  if(!db) throw new Error("Supabase is not configured yet.");
  const key=await sha256(code.trim());
  const ok=await rpc("hdg_check_group",{p_group_key:key});
  if(ok){
    groupKey=key;
    sessionStorage.setItem("hdg_group_key",key);
    return true;
  }
  return false;
}

  }
}
